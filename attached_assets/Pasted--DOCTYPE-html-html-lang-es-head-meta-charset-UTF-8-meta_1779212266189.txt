<script>
    // ---------- SEGURIDAD: solo acceso por contraseña (dueño) ----------
    let adminAuthorized = false;
    const OWNER_PASSWORD = "GTR2025Auto";  // contraseña de acceso al panel

    // Almacenamiento simulado (backend en memoria)
    let inventory = [
        { id: 1, name: "Aceite Motor Sintético 5W-30", price: 94.5, stock: 45, image: "https://placehold.co/400x300?text=Aceite+Motor" },
        { id: 2, name: "Filtro de Aire Deportivo", price: 28.9, stock: 22, image: "https://placehold.co/400x300?text=Filtro+Aire" },
        { id: 3, name: "Pastillas de Freno Cerámicas", price: 52.3, stock: 38, image: "https://placehold.co/400x300?text=Pastillas+Freno" },
        { id: 4, name: "Bujía Iridio (x4)", price: 41.2, stock: 10, image: "https://placehold.co/400x300?text=Bujias" }
    ];

    let salesOrders = [
        { id: 101, cliente: "Carlos M.", total: 94.5, fecha: "2025-04-01", estado: "Completado" },
        { id: 102, cliente: "Lucía G.", total: 81.2, fecha: "2025-04-02", estado: "Completado" }
    ];

    let campaigns = [
        { nombre: "Semana Santa", presupuesto: 500, segmentacion: "Automotriz", activa: true }
    ];

    let walletBalance = 12450.80;  // saldo disponible para transferir al dueño
    let socialAccounts = { instagram: "gtr_cubauto", facebook: "gtrcubauto", autoPost: false };

    let storeConfig = {
        bannerTitle: "REPUESTOS DE ALTA GAMA",
        primaryColor: "#e65c1e",
        showPromo: true,
        promoText: "10% OFF con transferencia"
    };

    // Helper para notificar
    function showNotif(msg, type = "info") {
        const notif = document.createElement("div");
        notif.className = "notification-toast";
        notif.innerHTML = `<i class="fas fa-bell"></i> ${msg}`;
        document.body.appendChild(notif);
        setTimeout(() => notif.remove(), 3000);
    }

    // Renderizar productos en tienda pública
    function renderStoreProducts() {
        const container = document.getElementById("productosTienda");
        if (!container) return;
        container.innerHTML = inventory.map(p => `
            <div class="product-card">
                <img src="${p.image}" alt="${p.name}">
                <h3>${p.name}</h3>
                <div class="price">$${p.price.toFixed(2)}</div>
                <button class="small-btn" onclick="addToCart(${p.id})"><i class="fas fa-cart-plus"></i> Comprar</button>
            </div>
        `).join('');
    }

    window.addToCart = function(productId) {
        // simulador de carrito y pago api (billetera integrada)
        const product = inventory.find(p => p.id === productId);
        if(!product) return;
        if(product.stock <=0) { showNotif("Sin stock", "error"); return; }
        // Simular pago desde billetera del dueño? Realmente se crea una orden y se acredita a la wallet.
        const amount = product.price;
        if(confirm(`Agregar ${product.name} por $${amount} ¿Deseas pagar con Wallet GTR? (saldo disponible: $${walletBalance.toFixed(2)})`)){
            // proceso de compra simulado - API de carrito
            walletBalance += amount;   // Dueño recibe el dinero (en realidad incrementa)
            product.stock -= 1;
            salesOrders.unshift({ id: Date.now(), cliente: "Cliente Web", total: amount, fecha: new Date().toISOString().slice(0,10), estado: "Pagado" });
            renderStoreProducts();
            showNotif(`Compra exitosa. Se agregó $${amount} a la billetera del dueño.`);
            if(adminAuthorized && document.getElementById("dynamicAdminContent")) renderAdminSection("dashboard");
        }
    };

    // Renderizar secciones del admin
    function renderAdminSection(section) {
        const adminContainer = document.getElementById("dynamicAdminContent");
        if(!adminContainer) return;
        switch(section) {
            case 'dashboard':
                adminContainer.innerHTML = `
                    <div class="metrics-grid">
                        <div class="metric-card"><div><i class="fas fa-chart-simple"></i> Ventas hoy</div><strong>$${(salesOrders.reduce((a,b)=>a+b.total,0)).toFixed(2)}</strong></div>
                        <div class="metric-card"><div><i class="fas fa-chart-line"></i> vs semana pasada</div><strong>+12%</strong></div>
                        <div class="metric-card"><div><i class="fas fa-box"></i> Stock total</div><strong>${inventory.reduce((a,b)=>a+b.stock,0)}</strong></div>
                    </div>
                    <h3>Productos más vendidos</h3>
                    <ul>${inventory.map(p=>`<li>${p.name} - $${p.price} (stock: ${p.stock})</li>`).join('')}</ul>
                    <h3>Órdenes recientes</h3>
                    <table class="data-table"><tr><th>ID</th><th>Cliente</th><th>Total</th><th>Estado</th></tr>
                    ${salesOrders.slice(0,5).map(o=>`<tr><td>${o.id}</td><td>${o.cliente}</td><td>$${o.total}</td><td><span class="badge">${o.estado}</span></td></tr>`).join('')}
                    </table>
                    <div class="campaign-form"><h4>Gestión campañas publicitarias</h4>
                    <input type="text" id="campName" placeholder="Nombre campaña"><input type="number" id="campBudget" placeholder="Presupuesto"><input id="campSegment" placeholder="Segmentación">
                    <button class="small-btn" onclick="createCampaign()">Crear campaña</button>
                    <div id="campaignsList">${campaigns.map(c=>`<p>📢 ${c.nombre} | $${c.presupuesto} | ${c.segmentacion}</p>`).join('')}</div>
                    </div>
                `;
                break;
            case 'inventario':
                adminContainer.innerHTML = `<h2>Gestión de Inventario</h2>
                <table class="data-table"><tr><th>ID</th><th>Nombre</th><th>Precio</th><th>Stock</th><th>Imagen</th><th>Acciones</th></tr>
                ${inventory.map(p=>`<tr><td>${p.id}</td><td><input type="text" id="name-${p.id}" value="${p.name}"></td><td><input type="number" id="price-${p.id}" value="${p.price}"></td><td><input type="number" id="stock-${p.id}" value="${p.stock}"></td><td><input type="text" id="img-${p.id}" value="${p.image}"></td><td><button class="small-btn" onclick="updateProduct(${p.id})">Actualizar</button></td></tr>`).join('')}
                </table>
                <button class="small-btn" onclick="addNewProduct()">➕ Nuevo producto</button>`;
                break;
            case 'wallet':
                adminContainer.innerHTML = `<div class="wallet-card"><h2>💰 Billetera Digital GTR</h2>
                <h1 style="color:var(--primary)">$${walletBalance.toFixed(2)} USD</h1>
                <button class="small-btn" id="transferToOwnerBtn"><i class="fas fa-exchange-alt"></i> Transferir saldo a cuenta del dueño</button>
                <p style="margin-top:1rem">Cada venta en tienda acredita automáticamente a esta wallet. Puedes retirar a cuenta bancaria.</p>
                </div>`;
                setTimeout(()=>{
                    document.getElementById("transferToOwnerBtn")?.addEventListener("click", ()=>{
                        if(walletBalance > 0){
                            alert(`Simulación de transferencia bancaria: $${walletBalance} enviados a la cuenta del dueño.`);
                            walletBalance = 0;
                            renderAdminSection('wallet');
                            showNotif("Transferencia completada con éxito (API bancaria simulada)");
                        } else alert("Saldo insuficiente");
                    });
                }, 50);
                break;
            case 'social':
                adminContainer.innerHTML = `<h2>Automatización de Redes Sociales</h2>
                <div class="campaign-form"><label>Instagram: </label><input id="igUser" value="${socialAccounts.instagram}"><label>Facebook: </label><input id="fbUser" value="${socialAccounts.facebook}"><label>Auto-publicación de ofertas</label><input type="checkbox" id="autoPostCB" ${socialAccounts.autoPost ? 'checked' : ''}>
                <button class="small-btn" onclick="saveSocialSettings()">Guardar y Vincular API</button>
                <p><i class="fab fa-instagram"></i> Programar publicación automática cada semana (simulación)</p>
                <button class="small-btn" onclick="simulateAutoPost()">Probar automatización</button>
                </div>`;
                break;
            case 'config':
                adminContainer.innerHTML = `<h2>Configuración General / Diseño</h2>
                <div class="campaign-form"><label>Título tienda:</label><input id="configStoreTitle" value="GTR CUBAUTO"><label>Color primario (naranja/rojo):</label><input type="color" id="primaryColorPicker" value="${storeConfig.primaryColor}"><label>¿Mostrar promo flotante?</label><input type="checkbox" id="showPromoCheck" ${storeConfig.showPromo ? 'checked':''}><label>Texto promoción:</label><input id="promoText" value="${storeConfig.promoText}">
                <button class="small-btn" onclick="saveGeneralConfig()">Aplicar cambios visuales</button>
                <hr><h3>Imágenes de productos destacados (subir demo)</h3><button class="small-btn" onclick="refreshStoreFromConfig()">Refrescar tienda</button></div>`;
                break;
            default: adminContainer.innerHTML = `<h3>Sección en construcción</h3><p>CRM / Analytics avanzado disponible pronto</p>`;
        }
    }

    // funciones helper admin CRUD
    window.updateProduct = (id) => {
        let product = inventory.find(p=>p.id===id);
        if(product){
            product.name = document.getElementById(`name-${id}`).value;
            product.price = parseFloat(document.getElementById(`price-${id}`).value);
            product.stock = parseInt(document.getElementById(`stock-${id}`).value);
            product.image = document.getElementById(`img-${id}`).value;
            renderStoreProducts();
            showNotif("Inventario actualizado + tienda sincronizada");
            renderAdminSection('inventario');
        }
    };
    window.addNewProduct = () => {
        let newId = Date.now();
        inventory.push({id:newId, name:"Nuevo repuesto", price:0, stock:0, image:"https://placehold.co/400x300?text=Auto+Part"});
        renderAdminSection('inventario');
    };
    window.createCampaign = () => {
        let name = document.getElementById("campName")?.value;
        let budget = document.getElementById("campBudget")?.value;
        let seg = document.getElementById("campSegment")?.value;
        if(name && budget) campaigns.push({nombre:name, presupuesto:+budget, segmentacion:seg, activa:true});
        renderAdminSection('dashboard');
        showNotif("Campaña creada con presupuesto desde API");
    };
    window.saveSocialSettings = () => {
        socialAccounts.instagram = document.getElementById("igUser")?.value;
        socialAccounts.facebook = document.getElementById("fbUser")?.value;
        socialAccounts.autoPost = document.getElementById("autoPostCB")?.checked;
        showNotif("Cuentas vinculadas (simulación API Meta/IG)");
    };
    window.simulateAutoPost = () => {
        showNotif("Automatización: Se publicó 'Ofertas del mes' en Instagram y Facebook.");
    };
    window.saveGeneralConfig = () => {
        storeConfig.primaryColor = document.getElementById("primaryColorPicker").value;
        storeConfig.showPromo = document.getElementById("showPromoCheck").checked;
        storeConfig.promoText = document.getElementById("promoText").value;
        document.documentElement.style.setProperty('--primary', storeConfig.primaryColor);
        renderStoreProducts();
        showNotif("Estilo & promociones actualizados en tiempo real");
    };
    window.refreshStoreFromConfig = () => { renderStoreProducts(); };

    // Autenticación: se muestra prompt al cargar para dueño
    function promptAdminAccess() {
        let pwd = prompt("🔐 Acceso exclusivo para el dueño de GTR CUBAUTO. Ingrese contraseña:");
        if(pwd === OWNER_PASSWORD){
            adminAuthorized = true;
            document.getElementById("storeView").style.display = "none";
            document.getElementById("adminPanel").style.display = "block";
            renderAdminSection("dashboard");
            // mostrar métricas en tiempo real
            setInterval(()=>{
                if(adminAuthorized) {
                    let now = new Date();
                    document.getElementById("liveDateTime") && (document.getElementById("liveDateTime").innerText = now.toLocaleDateString() + " " + now.toLocaleTimeString());
                }
            },1000);
        } else {
            if(pwd !== null) alert("Contraseña incorrecta. Acceso denegado.");
            adminAuthorized = false;
        }
    }
    document.getElementById("logoutBtn")?.addEventListener("click",()=>{
        adminAuthorized=false;
        document.getElementById("adminPanel").style.display = "none";
        document.getElementById("storeView").style.display = "block";
        showNotif("Sesión de administración cerrada con seguridad.");
    });
    document.getElementById("openOffersBtn")?.addEventListener("click", ()=>{
        alert("🔥 Ofertas especiales: 15% en filtros y aceites. Usa tu Wallet dentro del panel de administración.");
    });

    // Navegación menu admin
    document.getElementById("adminMenu")?.addEventListener("click", (e)=>{
        let target = e.target.closest("li");
        if(target && target.dataset.section){
            renderAdminSection(target.dataset.section);
        }
    });

    // inicializar tienda y evento acceso dueño (se muestra botón oculto o combinación)
    setTimeout(()=>{
        renderStoreProducts();
        // Agregar acceso rápido mediante tecla (Ctrl+Shift+A) o simplemente prompt al hacer doble click en logo
        const headerTitle = document.querySelector(".store-header h1");
        if(headerTitle){
            headerTitle.addEventListener("dblclick", ()=>{
                promptAdminAccess();
            });
        }
        // Hora inicial
        if(document.getElementById("liveDateTime")){
            setInterval(()=>{
                if(document.getElementById("liveDateTime"))
                    document.getElementById("liveDateTime").innerText = new Date().toLocaleString();
            },1000);
        }
    },100);
    // API extra para notificaciones de stock bajo
    setInterval(()=>{
